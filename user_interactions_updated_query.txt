SELECT 
    network_userid as user_id,
    listing_web_id as property_listing_id,
    
    -- STRENGTH OF PREFERENCE SCORE
    (
        (SUM(CASE WHEN event_name = 'content_view' THEN 1 ELSE 0 END) * 1) + 
        (SUM(CASE WHEN event_name = 'content_save' THEN 1 ELSE 0 END) * 5) +
        -- CRITICAL: This negative weight is useless if you filter the result later!
        (SUM(CASE WHEN event_name = 'content_unsave' THEN 1 ELSE 0 END) * -5) +
        (SUM(CASE WHEN event_name LIKE '%lead_click%' THEN 1 ELSE 0 END) * 10) +
        (SUM(CASE WHEN event_name LIKE '%lead_send%' OR event_name = 'instapage_lead' THEN 1 ELSE 0 END) * 50)
    ) as interaction_score,

    MAX(derived_timestamp) as last_interaction_time

FROM 
    pf_int_consumer_graph.stg_snowplow_events

WHERE 
    derived_timestamp >= DATEADD(month, -3, GETDATE()) 
    AND listing_web_id IS NOT NULL

GROUP BY 
    1, 2;
    
-- REMOVED: HAVING interaction_score > 0
-- REASON: We need negative/neutral signals for the model to learn what users REJECT.
