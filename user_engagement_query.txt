SELECT 
    listing_web_id as property_listing_id,
    
    -- METRICS (Keep these for analytics/debugging if needed)
    SUM(CASE WHEN event_name = 'content_view' THEN 1 ELSE 0 END) as view_count,
    SUM(CASE WHEN event_name = 'content_save' THEN 1 ELSE 0 END) as save_count,
    SUM(CASE WHEN event_name LIKE '%lead_click%' THEN 1 ELSE 0 END) as lead_click_count,
    SUM(CASE WHEN event_name LIKE '%lead_send%' THEN 1 ELSE 0 END) as lead_submission_count,
    
    -- --- WORLD CLASS SCORING (With Time Decay) ---
    SUM(
        (
            (CASE WHEN event_name = 'content_view' THEN 1.0 ELSE 0 END) +
            (CASE WHEN event_name = 'content_save' THEN 5.0 ELSE 0 END) +
            (CASE WHEN event_name IN ('lead_click', 'new_projects_lead_click', 'new_projects_dev_lead_click') THEN 10.0 ELSE 0 END) +
            (CASE WHEN event_name IN ('lead_send', 'new_projects_lead_send', 'new_projects_dev_lead_send', 'instapage_lead') THEN 50.0 ELSE 0 END)
        ) 
        * -- DECAY FACTOR: 0.98^Days_Ago
        -- Actions 30 days ago are worth ~54% of actions today.
        -- Actions 90 days ago are worth ~16% of actions today.
        POWER(0.98, DATEDIFF(day, derived_timestamp, GETDATE()))
    ) as popularity_score

FROM 
    pf_int_consumer_graph.stg_snowplow_events

WHERE 
    derived_timestamp >= DATEADD(day, -90, GETDATE()) 
    AND listing_web_id IS NOT NULL

GROUP BY 
    listing_web_id

-- OPTIMIZATION: Don't store properties with negligible scores
HAVING popularity_score > 1.0;
